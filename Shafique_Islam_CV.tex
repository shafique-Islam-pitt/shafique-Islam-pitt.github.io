\section*{Working Papers}
\paper{Delivering Degrees: Educational Spillovers from Amazon's Logistics Expansion.}
{Working paper. \href{https://shafique-Islam-pitt.github.io/manuscript.pdf}{[pdf copy]}}
{}

\begin{itemize}
  \item Estimates the causal effect of Amazon fulfillment center entry on local educational attainment using staggered entry timing across U.S. commuting zones from 2010 to 2023.
  \item Finds increases in bachelor's and master's degree attainment, with effects operating through household income gains and spillovers to information-intensive establishments and high-skill occupations.
  \item Winner, Best Second-Year Paper Award, Department of Economics, University of Pittsburgh (2026).
\end{itemize}

\paper{When Samples Shape Significance.}
{With Sonia Ale, Lester Lusher, Huda Osman, and Jacob Stenstrom.
\href{https://shafique-Islam-pitt.github.io/When_Samples_Shape_Significance.pdf}{[pdf copy]}}
{}

\begin{itemize}
  \item Studies how sampling variation shapes statistical significance by re-estimating results from top economics journals across independent resamples.
  \item Documents that statistical evidence can vary substantially even when the research design is held fixed.
\end{itemize}
