\section*{Working Papers}
\textbf{Warehouses and Diplomas: Amazon's Expansion and Local Educational Attainment}
\hfill \href{https://shafique-Islam-pitt.github.io/manuscript.pdf}{[PDF]}\par
\begin{itemize}
  \item Estimates the effect of Amazon's logistics facility rollout on local human capital,
    exploiting the staggered timing of Amazon's entry into U.S. commuting zones between 2010
    and 2024.
  \item Finds that Amazon's entry raised educational attainment, particularly bachelor's and
    master's degrees; more graduates moved to or stayed in treated areas, and students shifted
    from part-time to full-time enrollment.
  \item Shows that the effects reflect spillovers: Amazon's presence expanded skilled
    employment at other firms and raised local demand for college graduates.
\end{itemize}
\textit{Winner, Best Second-Year Paper Award, Department of Economics, University of Pittsburgh (2026).}
\vspace{0.5em}


\textbf{When Samples Shape Significance}
\hfill \href{https://shafique-Islam-pitt.github.io/When_Samples_Shape_Significance.pdf}{[PDF]}\par
\textit{with Sonia Ale, Lester Lusher, Huda Osman, and Jacob Stenstrom}
\begin{itemize}
  \item Replicates findings from 24 papers in top economics journals and re-estimates each
    result across 40 independent resamples drawn from the same data-generating process.
  \item Shows that sampling variation alone can overturn statistical significance even when
    the research design is held fixed.
\end{itemize}
